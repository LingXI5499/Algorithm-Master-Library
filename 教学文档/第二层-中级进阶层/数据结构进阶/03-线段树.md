# 03 - 线段树 (Segment Tree)

## 1. 概念解释

线段树是一棵完全二叉树，每个节点维护一段区间 `[l, r]` 的聚合信息（如区间和、最大值）。  
支持 **O(log n)** 的区间查询和单点/区间更新。

```
数组 a = [1, 3, 5, 7, 9, 11]  区间和线段树：
                 [0,5]=36
          [0,2]=9        [3,5]=27
       [0,1]=4  [2,2]=5  [3,4]=16  [5,5]=11
    [0,0]=1 [1,1]=3   [3,3]=7 [4,4]=9
```

## 2. 基本线段树（区间求和）

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 4e5 + 5;
long long tree[MAXN];

void build(vector<int>& a, int node, int l, int r) {
    if (l == r) { tree[node] = a[l]; return; }
    int mid = l + (r - l) / 2;
    build(a, 2*node, l, mid);
    build(a, 2*node+1, mid+1, r);
    tree[node] = tree[2*node] + tree[2*node+1];
}

// 单点更新：位置 pos 的值加 val
void update(int node, int l, int r, int pos, long long val) {
    if (l == r) { tree[node] += val; return; }
    int mid = l + (r - l) / 2;
    if (pos <= mid) update(2*node, l, mid, pos, val);
    else update(2*node+1, mid+1, r, pos, val);
    tree[node] = tree[2*node] + tree[2*node+1];
}

// 区间查询 [ql, qr] 的和
long long query(int node, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tree[node];
    int mid = l + (r - l) / 2;
    long long res = 0;
    if (ql <= mid) res += query(2*node, l, mid, ql, qr);
    if (qr > mid)  res += query(2*node+1, mid+1, r, ql, qr);
    return res;
}
```

**使用示例**：

```cpp
int main() {
    int n = 6;
    vector<int> a = {1, 3, 5, 7, 9, 11};
    build(a, 1, 0, n-1);
    cout << query(1, 0, n-1, 1, 3) << "\n";  // 3+5+7 = 15
    update(1, 0, n-1, 2, 10);                // a[2] += 10
    cout << query(1, 0, n-1, 1, 3) << "\n";  // 3+15+7 = 25
    return 0;
}
```

## 3. 懒惰标记（Lazy Propagation）

对区间进行批量修改时，用 **懒惰标记** 延迟更新子节点，避免逐个更新。

```cpp
long long tree[MAXN], lazy[MAXN];

void pushDown(int node, int l, int r) {
    if (lazy[node] == 0) return;
    int mid = l + (r - l) / 2;
    // 将懒惰标记传递给子节点
    tree[2*node]   += lazy[node] * (mid - l + 1);
    tree[2*node+1] += lazy[node] * (r - mid);
    lazy[2*node]   += lazy[node];
    lazy[2*node+1] += lazy[node];
    lazy[node] = 0;
}

// 区间 [ql, qr] 所有元素加 val
void rangeUpdate(int node, int l, int r, int ql, int qr, long long val) {
    if (ql <= l && r <= qr) {
        tree[node] += val * (r - l + 1);
        lazy[node] += val;
        return;
    }
    pushDown(node, l, r);
    int mid = l + (r - l) / 2;
    if (ql <= mid) rangeUpdate(2*node, l, mid, ql, qr, val);
    if (qr > mid)  rangeUpdate(2*node+1, mid+1, r, ql, qr, val);
    tree[node] = tree[2*node] + tree[2*node+1];
}

// 区间查询（同上，需先 pushDown）
long long query(int node, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return tree[node];
    pushDown(node, l, r);
    int mid = l + (r - l) / 2;
    long long res = 0;
    if (ql <= mid) res += query(2*node, l, mid, ql, qr);
    if (qr > mid)  res += query(2*node+1, mid+1, r, ql, qr);
    return res;
}
```

## 4. 区间最值线段树

```cpp
int treeMax[MAXN];

void buildMax(vector<int>& a, int node, int l, int r) {
    if (l == r) { treeMax[node] = a[l]; return; }
    int mid = l + (r - l) / 2;
    buildMax(a, 2*node, l, mid);
    buildMax(a, 2*node+1, mid+1, r);
    treeMax[node] = max(treeMax[2*node], treeMax[2*node+1]);
}

int queryMax(int node, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) return treeMax[node];
    int mid = l + (r - l) / 2;
    int res = INT_MIN;
    if (ql <= mid) res = max(res, queryMax(2*node, l, mid, ql, qr));
    if (qr > mid)  res = max(res, queryMax(2*node+1, mid+1, r, ql, qr));
    return res;
}
```

## 5. 动态开点线段树

当坐标范围极大（如 10⁹）时，按需创建节点：

```cpp
struct Node {
    long long sum, lazy;
    int left, right;  // 子节点编号
} seg[MAXN * 40];
int cnt = 0, root = 0;

int newNode() { cnt++; seg[cnt] = {0, 0, 0, 0}; return cnt; }

void push_down(int node, int l, int r) {
    if (!seg[node].lazy) return;
    if (!seg[node].left)  seg[node].left  = newNode();
    if (!seg[node].right) seg[node].right = newNode();
    int mid = l + (r - l) / 2;
    long long lz = seg[node].lazy;
    seg[seg[node].left].sum  += lz * (mid - l + 1);
    seg[seg[node].right].sum += lz * (r - mid);
    seg[seg[node].left].lazy  += lz;
    seg[seg[node].right].lazy += lz;
    seg[node].lazy = 0;
}
```

## 6. 例题：区间求和（LeetCode 307）

```cpp
class NumArray {
    int n;
    vector<long long> tree;
    void update_(int node, int l, int r, int pos, int val) {
        if (l == r) { tree[node] = val; return; }
        int mid = l + (r - l) / 2;
        if (pos <= mid) update_(2*node, l, mid, pos, val);
        else update_(2*node+1, mid+1, r, pos, val);
        tree[node] = tree[2*node] + tree[2*node+1];
    }
    long long query_(int node, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tree[node];
        int mid = l + (r - l) / 2; long long res = 0;
        if (ql <= mid) res += query_(2*node, l, mid, ql, qr);
        if (qr > mid)  res += query_(2*node+1, mid+1, r, ql, qr);
        return res;
    }
public:
    NumArray(vector<int>& nums) : n(nums.size()), tree(4*nums.size(), 0) {
        function<void(int,int,int)> build = [&](int node, int l, int r) {
            if (l == r) { tree[node] = nums[l]; return; }
            int mid = l+(r-l)/2;
            build(2*node, l, mid); build(2*node+1, mid+1, r);
            tree[node] = tree[2*node] + tree[2*node+1];
        };
        build(1, 0, n-1);
    }
    void update(int i, int val) { update_(1, 0, n-1, i, val); }
    int sumRange(int l, int r) { return query_(1, 0, n-1, l, r); }
};
```

## 7. 总结

| 操作 | 时间复杂度 |
|------|-----------|
| 建树 | O(n) |
| 单点更新 | O(log n) |
| 区间更新（lazy）| O(log n) |
| 区间查询 | O(log n) |

**线段树 vs 树状数组**：

| | 树状数组 | 线段树 |
|--|---------|-------|
| 代码量 | 少 | 多 |
| 常数 | 小 | 大 |
| 功能 | 前缀和/差分 | 任意区间聚合 |
| 区间修改 | 需差分技巧 | lazy 标记 |
