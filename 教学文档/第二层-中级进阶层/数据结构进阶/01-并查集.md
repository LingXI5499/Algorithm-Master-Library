# 01 - 并查集 (Union-Find)

## 1. 概念解释

### 什么是并查集？

并查集（Union-Find / Disjoint Set Union）是一种维护**元素分组**的数据结构，支持两种操作：
- **Find**：查找元素所属的集合（代表元）
- **Union**：合并两个集合

**核心应用**：判断连通性、连通分量计数、最小生成树（Kruskal）

---

## 2. 基础实现

```cpp
const int MAXN = 1e5 + 5;
int parent[MAXN];

void init(int n) {
    for (int i = 0; i <= n; i++) parent[i] = i;
}

int find(int x) {
    if (parent[x] != x) parent[x] = find(parent[x]);  // 路径压缩
    return parent[x];
}

void unite(int x, int y) {
    parent[find(x)] = find(y);
}

bool connected(int x, int y) {
    return find(x) == find(y);
}
```

---

## 3. 优化：按秩合并 + 路径压缩

```cpp
int parent[MAXN], rank_[MAXN];

void init(int n) {
    for (int i = 0; i <= n; i++) { parent[i] = i; rank_[i] = 0; }
}

int find(int x) {
    if (parent[x] != x) parent[x] = find(parent[x]);
    return parent[x];
}

void unite(int x, int y) {
    int rx = find(x), ry = find(y);
    if (rx == ry) return;
    // 秩小的接到秩大的下面，保持树矮
    if (rank_[rx] < rank_[ry]) swap(rx, ry);
    parent[ry] = rx;
    if (rank_[rx] == rank_[ry]) rank_[rx]++;
}
```

> 路径压缩 + 按秩合并后，Find 和 Union 的均摊时间复杂度为 **O(α(n))**，实际上接近 O(1)。

## 4. 带权并查集

用于维护元素间的相对权值关系（如差值、倍数）。

```cpp
// 带权并查集：weight[x] 表示 x 到 parent[x] 的权值
int parent[MAXN];
long long weight[MAXN];  // 到父节点的距离

void init(int n) {
    for (int i = 0; i <= n; i++) { parent[i] = i; weight[i] = 0; }
}

// 返回 x 的根，同时更新 weight[x] 为 x 到根的总权值
int find(int x) {
    if (parent[x] == x) return x;
    int root = find(parent[x]);
    weight[x] += weight[parent[x]];  // 累积权值
    parent[x] = root;
    return root;
}

// 合并 x 和 y，且 weight(x) - weight(y) = w
void unite(int x, int y, long long w) {
    int rx = find(x), ry = find(y);
    if (rx == ry) return;
    parent[rx] = ry;
    weight[rx] = weight[y] - weight[x] + w;
}
```

## 5. 经典例题

### 例题 1：岛屿数量（并查集版，LeetCode 200）

```cpp
class UnionFind {
    vector<int> p;
    int cnt;
public:
    UnionFind(int n) : p(n), cnt(0) { iota(p.begin(), p.end(), 0); }
    int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }
    void unite(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx != ry) { p[rx] = ry; cnt--; }
    }
    void add() { cnt++; }
    int count() { return cnt; }
};

int numIslands(vector<vector<char>>& grid) {
    int n = grid.size(), m = grid[0].size();
    UnionFind uf(n * m);
    int dx[] = {0, 1}, dy[] = {1, 0};
    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            if (grid[i][j] == '1') {
                uf.add();
                for (int d = 0; d < 2; d++) {
                    int ni = i + dx[d], nj = j + dy[d];
                    if (ni < n && nj < m && grid[ni][nj] == '1')
                        uf.unite(i*m+j, ni*m+nj);
                }
            }
    return uf.count();
}
```

### 例题 2：账户合并（LeetCode 721）

```cpp
vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
    unordered_map<string, int> emailId;
    vector<int> parent;
    int id = 0;

    function<int(int)> find = [&](int x) -> int {
        return parent[x] == x ? x : parent[x] = find(parent[x]);
    };
    auto unite = [&](int x, int y) {
        parent[find(x)] = find(y);
    };

    for (auto& acc : accounts)
        for (int i = 1; i < (int)acc.size(); i++) {
            if (!emailId.count(acc[i])) { emailId[acc[i]] = id++; parent.push_back(id-1); }
            unite(emailId[acc[1]], emailId[acc[i]]);
        }

    unordered_map<int, vector<string>> groups;
    for (auto& [email, eid] : emailId)
        groups[find(eid)].push_back(email);

    // 找每个 root 对应的账户名
    unordered_map<int, string> rootName;
    for (auto& acc : accounts)
        for (int i = 1; i < (int)acc.size(); i++)
            rootName[find(emailId[acc[i]])] = acc[0];

    vector<vector<string>> res;
    for (auto& [root, emails] : groups) {
        sort(emails.begin(), emails.end());
        emails.insert(emails.begin(), rootName[root]);
        res.push_back(emails);
    }
    return res;
}
```

### 例题 3：冗余连接（LeetCode 684）

找有向图中第一条形成环的边：

```cpp
vector<int> findRedundantConnection(vector<vector<int>>& edges) {
    int n = edges.size();
    vector<int> p(n + 1);
    iota(p.begin(), p.end(), 0);
    function<int(int)> find = [&](int x) -> int {
        return p[x] == x ? x : p[x] = find(p[x]);
    };
    for (auto& e : edges) {
        int u = find(e[0]), v = find(e[1]);
        if (u == v) return e;
        p[u] = v;
    }
    return {};
}
```

## 6. 总结

| 优化 | 效果 |
|------|------|
| 路径压缩 | Find 均摊接近 O(1) |
| 按秩合并 | 防止树退化为链 |
| 两者结合 | 均摊 O(α(n))，实际 O(1) |

**适用场景**：连通性查询、动态合并集合、最小生成树（Kruskal）、离线最近公共祖先
