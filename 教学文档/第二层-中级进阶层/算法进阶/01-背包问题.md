# 01 - 背包问题

## 1. 概念解释

背包问题是一类经典的**动态规划**问题：给定若干物品和一个容量为 W 的背包，在不超过容量的前提下，使选取物品的总价值最大。

---

## 2. 0-1 背包

每件物品**最多选一次**。

**状态定义**：`dp[j]` = 容量为 j 时的最大价值  
**转移**：`dp[j] = max(dp[j], dp[j - w[i]] + v[i])`  
**遍历顺序**：容量 j **从大到小**（防止一件物品被重复选）

```cpp
#include <bits/stdc++.h>
using namespace std;

int zeroOnePack(int W, vector<int>& w, vector<int>& v) {
    int n = w.size();
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < n; i++)
        for (int j = W; j >= w[i]; j--)
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
    return dp[W];
}
```

### 例题：分割等和子集（LeetCode 416）

判断能否将数组分成两个和相等的子集（即：能否选一些数，和恰好为 sum/2）。

```cpp
bool canPartition(vector<int>& nums) {
    int sum = accumulate(nums.begin(), nums.end(), 0);
    if (sum % 2 != 0) return false;
    int target = sum / 2;
    vector<bool> dp(target + 1, false);
    dp[0] = true;
    for (int x : nums)
        for (int j = target; j >= x; j--)
            dp[j] = dp[j] || dp[j - x];
    return dp[target];
}
```

## 3. 完全背包

每件物品**可以选无限次**。

**转移**：`dp[j] = max(dp[j], dp[j - w[i]] + v[i])`  
**遍历顺序**：容量 j **从小到大**（允许重复选同一件）

```cpp
int completePack(int W, vector<int>& w, vector<int>& v) {
    int n = w.size();
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < n; i++)
        for (int j = w[i]; j <= W; j++)     // 正向遍历
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
    return dp[W];
}
```

### 例题：零钱兑换（LeetCode 322）

```cpp
int coinChange(vector<int>& coins, int amount) {
    vector<int> dp(amount + 1, INT_MAX);
    dp[0] = 0;
    for (int c : coins)
        for (int j = c; j <= amount; j++)
            if (dp[j - c] != INT_MAX)
                dp[j] = min(dp[j], dp[j - c] + 1);
    return dp[amount] == INT_MAX ? -1 : dp[amount];
}
```

### 例题：零钱兑换 II（LeetCode 518，方案数）

```cpp
int change(int amount, vector<int>& coins) {
    vector<long long> dp(amount + 1, 0);
    dp[0] = 1;
    for (int c : coins)
        for (int j = c; j <= amount; j++)
            dp[j] += dp[j - c];
    return dp[amount];
}
```

## 4. 多重背包

每件物品最多选 `k[i]` 次。

### 朴素 O(n × W × k)

```cpp
int multiPack(int W, vector<int>& w, vector<int>& v, vector<int>& k) {
    int n = w.size();
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < n; i++)
        for (int j = W; j >= 0; j--)
            for (int t = 1; t <= k[i] && t * w[i] <= j; t++)
                dp[j] = max(dp[j], dp[j - t*w[i]] + t*v[i]);
    return dp[W];
}
```

### 二进制拆分优化 O(n × W × log k)

将 k[i] 个同种物品拆成 1, 2, 4, ..., 2^m, 余数 等组，转为 0-1 背包：

```cpp
int multiPackBinary(int W, vector<int>& w, vector<int>& v, vector<int>& k) {
    vector<int> nw, nv;
    for (int i = 0; i < (int)w.size(); i++) {
        int rem = k[i];
        for (int p = 1; p <= rem; p <<= 1) {
            nw.push_back(p * w[i]);
            nv.push_back(p * v[i]);
            rem -= p;
        }
        if (rem > 0) { nw.push_back(rem * w[i]); nv.push_back(rem * v[i]); }
    }
    // 转为 0-1 背包
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < (int)nw.size(); i++)
        for (int j = W; j >= nw[i]; j--)
            dp[j] = max(dp[j], dp[j - nw[i]] + nv[i]);
    return dp[W];
}
```

## 5. 分组背包

物品分为若干组，每组**最多选一件**。

```cpp
int groupPack(int W, vector<vector<pair<int,int>>>& groups) {
    // groups[i] = [(w,v), ...] 第 i 组的物品
    vector<int> dp(W + 1, 0);
    for (auto& group : groups)
        for (int j = W; j >= 0; j--)          // 倒序（同 0-1 背包）
            for (auto& [wi, vi] : group)
                if (j >= wi)
                    dp[j] = max(dp[j], dp[j - wi] + vi);
    return dp[W];
}
```

## 6. 二维背包

背包有两种容量限制（如重量和体积）。

```cpp
int twoDimPack(int W, int V, vector<int>& w, vector<int>& vol, vector<int>& v) {
    int n = w.size();
    vector<vector<int>> dp(W + 1, vector<int>(V + 1, 0));
    for (int i = 0; i < n; i++)
        for (int j = W; j >= w[i]; j--)
            for (int k = V; k >= vol[i]; k--)
                dp[j][k] = max(dp[j][k], dp[j-w[i]][k-vol[i]] + v[i]);
    return dp[W][V];
}
```

**例题：一和零（LeetCode 474）**

```cpp
int findMaxForm(vector<string>& strs, int m, int n) {
    vector<vector<int>> dp(m+1, vector<int>(n+1, 0));
    for (auto& s : strs) {
        int zeros = count(s.begin(), s.end(), '0');
        int ones  = s.size() - zeros;
        for (int j = m; j >= zeros; j--)
            for (int k = n; k >= ones; k--)
                dp[j][k] = max(dp[j][k], dp[j-zeros][k-ones] + 1);
    }
    return dp[m][n];
}
```

## 7. 背包求方案数

将最大价值改为**方案计数**：

```cpp
int countWays(int W, vector<int>& w) {
    vector<long long> dp(W + 1, 0);
    dp[0] = 1;
    for (int wi : w)
        for (int j = W; j >= wi; j--)
            dp[j] += dp[j - wi];
    return dp[W];
}
```

## 8. 总结

| 类型 | 每件可选次数 | j 遍历方向 |
|------|------------|-----------|
| 0-1 背包 | 1 次 | 从大到小 |
| 完全背包 | 无限次 | 从小到大 |
| 多重背包 | k[i] 次 | 二进制拆分 |
| 分组背包 | 组内选 1 件 | 从大到小 |

> **记忆技巧**：0-1 背包倒序是为了防止"同一件物品被选多次"；完全背包正序是为了"允许重复选"。
