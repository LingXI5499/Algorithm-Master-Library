# 02 - 图论算法

## 1. 图的表示

### 邻接表（推荐，稀疏图）

```cpp
const int MAXN = 1e5 + 5;
vector<pair<int,int>> adj[MAXN];  // adj[u] = [(v, w), ...]

void addEdge(int u, int v, int w) {
    adj[u].push_back({v, w});
    // adj[v].push_back({u, w});  // 无向图加这行
}
```

### 链式前向星（竞赛常用，速度快）

```cpp
struct Edge { int to, next, w; } edges[MAXN * 2];
int head[MAXN], tot = 0;

void addEdge(int u, int v, int w) {
    edges[++tot] = {v, head[u], w};
    head[u] = tot;
}

// 遍历 u 的所有邻接边
for (int i = head[u]; i; i = edges[i].next) {
    int v = edges[i].to, w = edges[i].w;
}
```

## 2. 最短路算法

### Dijkstra（非负权有向/无向图）

时间复杂度：O((V + E) log V)

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef pair<long long, int> pli;

const long long INF = 1e18;

vector<long long> dijkstra(int src, int n, vector<vector<pair<int,int>>>& adj) {
    vector<long long> dist(n + 1, INF);
    priority_queue<pli, vector<pli>, greater<pli>> pq;
    dist[src] = 0;
    pq.push({0, src});

    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;  // 已有更短路，跳过
        for (auto [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}
```

### Bellman-Ford（允许负权，可检测负环）

时间复杂度：O(VE)

```cpp
struct Edge { int u, v, w; };

vector<int> bellmanFord(int src, int n, vector<Edge>& edges) {
    vector<int> dist(n + 1, INT_MAX);
    dist[src] = 0;
    for (int i = 0; i < n - 1; i++) {  // 松弛 n-1 次
        for (auto& e : edges) {
            if (dist[e.u] != INT_MAX && dist[e.u] + e.w < dist[e.v])
                dist[e.v] = dist[e.u] + e.w;
        }
    }
    // 检测负环（第 n 次仍可松弛）
    for (auto& e : edges)
        if (dist[e.u] != INT_MAX && dist[e.u] + e.w < dist[e.v])
            return {};  // 存在负环
    return dist;
}
```

### SPFA（Bellman-Ford 队列优化，最坏 O(VE)）

```cpp
vector<int> spfa(int src, int n, vector<vector<pair<int,int>>>& adj) {
    vector<int> dist(n + 1, INT_MAX);
    vector<bool> inQueue(n + 1, false);
    queue<int> q;
    dist[src] = 0;
    q.push(src); inQueue[src] = true;

    while (!q.empty()) {
        int u = q.front(); q.pop();
        inQueue[u] = false;
        for (auto [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                if (!inQueue[v]) { q.push(v); inQueue[v] = true; }
            }
        }
    }
    return dist;
}
```

### Floyd（多源最短路，O(V³)）

```cpp
void floyd(vector<vector<long long>>& dist, int n) {
    // 初始化：dist[i][j] = 边权，无边为 INF，dist[i][i] = 0
    for (int k = 1; k <= n; k++)
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++)
                if (dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];
}
```

## 3. 最小生成树

### Kruskal（边排序 + 并查集）

时间复杂度：O(E log E)

```cpp
struct Edge { int u, v, w; };
int parent[MAXN];
int find(int x) { return parent[x] == x ? x : parent[x] = find(parent[x]); }

long long kruskal(int n, vector<Edge>& edges) {
    sort(edges.begin(), edges.end(), [](auto& a, auto& b){ return a.w < b.w; });
    for (int i = 1; i <= n; i++) parent[i] = i;
    long long mstCost = 0;
    int edgesUsed = 0;
    for (auto& e : edges) {
        int ru = find(e.u), rv = find(e.v);
        if (ru != rv) {
            parent[ru] = rv;
            mstCost += e.w;
            if (++edgesUsed == n - 1) break;
        }
    }
    return mstCost;
}
```

### Prim（适合稠密图）

时间复杂度：O(E log V)（堆优化）

```cpp
long long prim(int src, int n, vector<vector<pair<int,int>>>& adj) {
    vector<long long> key(n + 1, LLONG_MAX);
    vector<bool> inMST(n + 1, false);
    priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
    key[src] = 0;
    pq.push({0, src});
    long long cost = 0;
    int cnt = 0;

    while (!pq.empty() && cnt < n) {
        auto [d, u] = pq.top(); pq.pop();
        if (inMST[u]) continue;
        inMST[u] = true; cost += d; cnt++;
        for (auto [v, w] : adj[u])
            if (!inMST[v] && w < key[v]) {
                key[v] = w; pq.push({key[v], v});
            }
    }
    return cost;
}
```

## 4. 拓扑排序

### Kahn 算法（BFS）

```cpp
vector<int> topoSort(int n, vector<vector<int>>& adj) {
    vector<int> inDeg(n + 1, 0);
    for (int u = 1; u <= n; u++)
        for (int v : adj[u]) inDeg[v]++;

    queue<int> q;
    for (int i = 1; i <= n; i++)
        if (inDeg[i] == 0) q.push(i);

    vector<int> order;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : adj[u])
            if (--inDeg[v] == 0) q.push(v);
    }
    return order.size() == n ? order : vector<int>{};  // 空表示有环
}
```

## 5. 强连通分量（Tarjan 算法）

```cpp
int dfn[MAXN], low[MAXN], sccId[MAXN];
bool onStack[MAXN];
stack<int> st;
int timer_ = 0, sccCnt = 0;

void tarjan(int u, vector<vector<int>>& adj) {
    dfn[u] = low[u] = ++timer_;
    st.push(u); onStack[u] = true;
    for (int v : adj[u]) {
        if (!dfn[v]) { tarjan(v, adj); low[u] = min(low[u], low[v]); }
        else if (onStack[v]) low[u] = min(low[u], dfn[v]);
    }
    if (low[u] == dfn[u]) {  // u 是 SCC 的根
        sccCnt++;
        while (true) {
            int v = st.top(); st.pop();
            onStack[v] = false;
            sccId[v] = sccCnt;
            if (v == u) break;
        }
    }
}
```

## 6. 例题：网络延迟时间（LeetCode 743）

从节点 k 出发，最短到达所有节点的时间：

```cpp
int networkDelayTime(vector<vector<int>>& times, int n, int k) {
    vector<vector<pair<int,int>>> adj(n + 1);
    for (auto& t : times) adj[t[0]].push_back({t[1], t[2]});

    const long long INF = 1e18;
    vector<long long> dist(n + 1, INF);
    priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
    dist[k] = 0; pq.push({0, k});

    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;
        for (auto [v, w] : adj[u])
            if (dist[u] + w < dist[v]) { dist[v] = dist[u] + w; pq.push({dist[v], v}); }
    }

    long long ans = *max_element(dist.begin() + 1, dist.end());
    return ans == INF ? -1 : ans;
}
```

## 7. 总结

| 算法 | 适用场景 | 时间复杂度 |
|------|----------|-----------|
| Dijkstra | 单源最短路（非负权）| O((V+E) log V) |
| Bellman-Ford | 单源最短路（含负权）| O(VE) |
| SPFA | Bellman-Ford 优化 | 均摊 O(E)，最坏 O(VE) |
| Floyd | 多源最短路 | O(V³) |
| Kruskal | 最小生成树（稀疏图）| O(E log E) |
| Prim | 最小生成树（稠密图）| O(E log V) |
| Kahn | 拓扑排序 | O(V+E) |
| Tarjan | 强连通分量 | O(V+E) |
