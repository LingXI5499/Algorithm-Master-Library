# 02 - 排序算法

## 1. 排序算法总览

| 算法 | 平均时间 | 最坏时间 | 空间 | 稳定性 |
|------|----------|----------|------|--------|
| 冒泡 | O(n²) | O(n²) | O(1) | 稳定 |
| 选择 | O(n²) | O(n²) | O(1) | 不稳定 |
| 插入 | O(n²) | O(n²) | O(1) | 稳定 |
| 归并 | O(n log n) | O(n log n) | O(n) | 稳定 |
| 快速 | O(n log n) | O(n²) | O(log n) | 不稳定 |
| 堆排 | O(n log n) | O(n log n) | O(1) | 不稳定 |
| 计数 | O(n+k) | O(n+k) | O(k) | 稳定 |
| 桶排 | O(n+k) | O(n²) | O(n+k) | 稳定 |
| 基数 | O(d·(n+k)) | O(d·(n+k)) | O(n+k) | 稳定 |

## 2. O(n²) 基础排序

### 冒泡排序

每轮将最大元素"冒泡"到末尾，加入提前退出优化：

```cpp
void bubbleSort(vector<int>& a) {
    int n = a.size();
    for (int i = 0; i < n - 1; i++) {
        bool swapped = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j+1]) {
                swap(a[j], a[j+1]);
                swapped = true;
            }
        }
        if (!swapped) break;  // 已有序，提前退出
    }
}
```

### 选择排序

每轮找最小值放到已排序区的末尾：

```cpp
void selectionSort(vector<int>& a) {
    int n = a.size();
    for (int i = 0; i < n - 1; i++) {
        int minIdx = i;
        for (int j = i + 1; j < n; j++)
            if (a[j] < a[minIdx]) minIdx = j;
        swap(a[i], a[minIdx]);
    }
}
```

### 插入排序

将每个元素插入到已排序区的合适位置（类似整理扑克牌）：

```cpp
void insertionSort(vector<int>& a) {
    int n = a.size();
    for (int i = 1; i < n; i++) {
        int key = a[i], j = i - 1;
        while (j >= 0 && a[j] > key) {
            a[j+1] = a[j];
            j--;
        }
        a[j+1] = key;
    }
}
```

> 插入排序在**近乎有序**的数据上表现极好，接近 O(n)。

## 3. O(n log n) 高效排序

### 归并排序（稳定，分治）

```cpp
void mergeSort(vector<int>& a, int lo, int hi) {
    if (hi - lo < 2) return;
    int mid = lo + (hi - lo) / 2;
    mergeSort(a, lo, mid);
    mergeSort(a, mid, hi);
    // 合并两个有序子数组
    vector<int> tmp(a.begin() + lo, a.begin() + hi);
    int i = 0, j = mid - lo, k = lo;
    int llen = mid - lo, rlen = hi - mid;
    while (i < llen && j < llen + rlen)
        a[k++] = tmp[i] <= tmp[j] ? tmp[i++] : tmp[j++];
    while (i < llen) a[k++] = tmp[i++];
    while (j < llen + rlen) a[k++] = tmp[j++];
}
```

### 快速排序（原地，随机化）

```cpp
int partition(vector<int>& a, int lo, int hi) {
    // 随机选取枢轴，避免最坏 O(n²)
    int randIdx = lo + rand() % (hi - lo + 1);
    swap(a[randIdx], a[hi]);
    int pivot = a[hi], i = lo - 1;
    for (int j = lo; j < hi; j++)
        if (a[j] <= pivot) swap(a[++i], a[j]);
    swap(a[i+1], a[hi]);
    return i + 1;
}

void quickSort(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return;
    int p = partition(a, lo, hi);
    quickSort(a, lo, p - 1);
    quickSort(a, p + 1, hi);
}
```

**三路快排**（处理大量重复元素）：

```cpp
void quickSort3Way(vector<int>& a, int lo, int hi) {
    if (lo >= hi) return;
    int pivot = a[lo + rand() % (hi - lo + 1)];
    int lt = lo, gt = hi, i = lo;
    // [lo..lt-1] < pivot  [lt..gt] == pivot  [gt+1..hi] > pivot
    while (i <= gt) {
        if      (a[i] < pivot) swap(a[lt++], a[i++]);
        else if (a[i] > pivot) swap(a[i], a[gt--]);
        else                   i++;
    }
    quickSort3Way(a, lo, lt - 1);
    quickSort3Way(a, gt + 1, hi);
}
```

### 堆排序

```cpp
void heapSort(vector<int>& a) {
    int n = a.size();
    auto down = [&](int i, int sz) {
        while (2*i+1 < sz) {
            int j = 2*i+1;
            if (j+1 < sz && a[j+1] > a[j]) j++;
            if (a[i] >= a[j]) break;
            swap(a[i], a[j]); i = j;
        }
    };
    // 建堆
    for (int i = n/2 - 1; i >= 0; i--) down(i, n);
    // 排序
    for (int i = n-1; i > 0; i--) {
        swap(a[0], a[i]);
        down(0, i);
    }
}
```

## 4. 线性时间排序

### 计数排序（适合小范围整数）

```cpp
void countingSort(vector<int>& a, int maxVal) {
    vector<int> cnt(maxVal + 1, 0);
    for (int x : a) cnt[x]++;
    int idx = 0;
    for (int v = 0; v <= maxVal; v++)
        while (cnt[v]--) a[idx++] = v;
}
```

### 基数排序（按位排序，适合大整数）

```cpp
void radixSort(vector<int>& a) {
    int maxVal = *max_element(a.begin(), a.end());
    for (int exp = 1; maxVal / exp > 0; exp *= 10) {
        int n = a.size();
        vector<int> out(n);
        vector<int> cnt(10, 0);
        for (int x : a) cnt[(x / exp) % 10]++;
        for (int i = 1; i < 10; i++) cnt[i] += cnt[i-1];
        for (int i = n-1; i >= 0; i--) {
            out[cnt[(a[i]/exp)%10] - 1] = a[i];
            cnt[(a[i]/exp)%10]--;
        }
        a = out;
    }
}
```

## 5. STL 排序

```cpp
#include <algorithm>
vector<int> a = {5, 3, 1, 4, 2};

// 升序
sort(a.begin(), a.end());

// 降序
sort(a.begin(), a.end(), greater<int>());

// 自定义比较
sort(a.begin(), a.end(), [](int x, int y){ return x > y; });

// 稳定排序（保留相等元素的相对顺序）
stable_sort(a.begin(), a.end());

// 部分排序（找前 k 小）
partial_sort(a.begin(), a.begin() + k, a.end());

// nth_element：使第 k 个位置上的元素为排好序时的值，O(n) 均摊
nth_element(a.begin(), a.begin() + k, a.end());
```

## 6. 总结

- 竞赛中直接使用 `sort()`，背后是**IntroSort**（快排+堆排+插入排序的混合）
- 数据量小（n < 50）：插入排序
- 需要稳定：归并排序
- 内存受限：堆排序或快速排序
- 数据范围小的整数：计数排序/基数排序
